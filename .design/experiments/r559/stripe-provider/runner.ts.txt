type Fixture=Awaited<ReturnType<typeof fixture>>;

 return list.data.map(i=>{test(i);return{id:i.id,status:i.status,total:i.total,paid:i.amount_paid,due:i.amount_due,reason:i.billing_reason,created:i.created,lines:i.lines.data.map(l=>({amount:l.amount,proration:l.parent?.subscription_item_details?.proration??null,period:l.period}))};});

async function advance(f:Fixture,to:number){
 assert.ok(to>(await s.testHelpers.testClocks.retrieve(f.clock.id)).frozen_time);
 test(await s.testHelpers.testClocks.advance(f.clock.id,{frozen_time:to}));
 const deadline=Date.now()+90000;
 for(;;){const clock=test(await s.testHelpers.testClocks.retrieve(f.clock.id));if(clock.status==='ready')return;assert.notEqual(clock.status,'internal_failure');if(Date.now()>deadline)throw Error('TEST_CLOCK_TIMEOUT');await sleep(1500);}
}
async function pause(f:Fixture,sub:Stripe.Subscription):Promise<PauseOperation>{
 test(sub);const plan=await f.provider.plan(f.parent,f.card);assert.ok(plan,'provider must accept supported plan');
 const result=pauseQuote({weeks:2,used:0,now:f.base*1000,zone:'America/Vancouver',book:{id:randomUUID(),number:1,shippedAt:new Date((f.base-3*86400)*1000).toISOString(),deliveredAt:new Date((f.base-86400)*1000).toISOString()},plan});assert.equal(result.ok,true);if(!result.ok)throw Error(result.error);
 const op:PauseOperation={id:randomUUID(),state:'open',send:true,quote:result.quote};
 const outcome=await f.provider.apply(f.parent,f.card,op);assert.equal(outcome,'applied',`provider pause outcome ${outcome}`);
 assert.equal(await f.provider.recover(f.parent,f.card,op),true);
 log('pause_applied',{kind:plan.kind,subscription:sub.id,previous:op.quote.previousCharge,resume:op.quote.chargeAt});return op;
}
async function initial(f:Fixture){const result=await f.start();assert.equal(result.ok,true);if(!result.ok)throw Error(result.error);const sub=test(await s.subscriptions.retrieve(result.subscription));report.resources.push({kind:'price',id:sub.items.data[0].price.id});save();assert.equal(sub.status,'trialing');return sub;}
async function paid(f:Fixture){const old=await initial(f);await stopPlanNow(s,old.id);assert.equal((await s.subscriptions.retrieve(old.id)).status,'canceled');const result=await f.start('restart');assert.equal(result.ok,true);if(!result.ok)throw Error(result.error);const sub=test(await s.subscriptions.retrieve(result.subscription));report.resources.push({kind:'price',id:sub.items.data[0].price.id});save();assert.equal(sub.status,'active');assert.equal(sub.trial_end,null);assert.equal(sub.billing_mode.type,'flexible');return sub;}
