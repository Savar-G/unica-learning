async function capture(name, screenshot = false) {
  const snap = await snapshot(qa); console.log(snap.diff || snap.tree);
  const geometry = await qa.evaluate(() => ({ width: innerWidth, scroll: document.documentElement.scrollWidth,
    text: document.body.innerText, shells: document.querySelectorAll('.pd-shell').length,
    alerts: [...document.querySelectorAll('[role="alert"]')].map(x => x.textContent),
    buttons: [...document.querySelectorAll('main button')].map(x => ({text:x.textContent,disabled:x.disabled,height:x.getBoundingClientRect().height})),
    requests: window.__qa.requests }));
  check(geometry.scroll <= geometry.width, name + ' horizontal overflow');
  check(geometry.shells === 1, name + ' duplicate frame');
  check(geometry.buttons.every(x => x.height >= 44), name + ' small button');
  cases.push({ name, ...geometry });
  if (screenshot && (name.endsWith('-390') || ['closed-1440','pause-early-1440','recovery-820'].includes(name))) {
    const bytes = await qa.screenshot({fullPage:false});
    console.log('PNG_EXPORT ' + name + ' ' + bytes.toString('base64'));
  }
  return geometry;
}
